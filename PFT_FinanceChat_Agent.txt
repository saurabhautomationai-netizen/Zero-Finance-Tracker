const rows = $input.all().map(i => i.json);
const questionRaw = $("Telegram Trigger").first().json.message.text || '';
const question = questionRaw.toLowerCase();

function amount(v) {
  return Number(String(v || 0).replace(/[^0-9.]/g, '')) || 0;
}

function norm(v) {
  return String(v || '').toLowerCase().trim();
}

const categoryMap = {
  food: ['food', 'zomato', 'swiggy', 'restaurant', 'dinner', 'lunch', 'snacks'],
  entertainment: ['entertainment', 'netflix', 'spotify', 'movie', 'subscription'],
  rent: ['rent', 'house rent', 'landlord'],
  health: ['health', 'medicine', 'medicines', 'apollo', 'pharmacy'],
  shopping: ['shopping', 'amazon', 'myntra', 'flipkart', 'clothes'],
  education: ['education', 'course', 'udemy', 'learning'],
  utilities: ['utilities', 'electricity', 'gas', 'bill', 'airtel', 'wifi', 'broadband'],
  transport: ['transport', 'uber', 'rapido', 'ola', 'taxi', 'travel'],
  salary: ['salary'],
  freelance: ['freelance', 'client', 'project'],
  cashback: ['cashback'],
  investment: ['investment', 'mutual fund', 'dividend'],
  miscellaneous: ['miscellaneous', 'misc', 'other']
};

const isIncome = /(income|earned|received|salary|credited|got paid|freelance|cashback|bonus)/i.test(question);
const isExpense = /(spent|expense|paid|bought|purchase|cost|bill|rent)/i.test(question);

let typeFilter = isIncome ? 'INCOME' : isExpense ? 'EXPENSE' : '';

let matchedCategory = '';
for (const [category, keywords] of Object.entries(categoryMap)) {
  if (keywords.some(k => question.includes(k))) {
    matchedCategory = category;
    break;
  }
}

let filtered = rows.filter(r => {
  const rowType = norm(r.Type);
  const rowCategory = norm(r.Category);
  const rowVendor = norm(r.Vendor);
  const rowProduct = norm(r.Product);

  if (typeFilter && rowType !== typeFilter.toLowerCase()) return false;

  if (matchedCategory) {
    const keywords = categoryMap[matchedCategory];
    return keywords.some(k =>
      rowCategory.includes(k) ||
      rowVendor.includes(k) ||
      rowProduct.includes(k)
    );
  }

  return true;
});

const total = filtered.reduce((sum, r) => sum + amount(r.Amount), 0);

return [{
  json: {
    question: questionRaw,
    typeFilter: typeFilter || 'ALL',
    matchedCategory: matchedCategory || 'ALL',
    total,
    count: filtered.length,
    answer: `Found ${filtered.length} matching transaction(s). Total: ₹${total.toLocaleString('en-IN')}`
  }
}];
