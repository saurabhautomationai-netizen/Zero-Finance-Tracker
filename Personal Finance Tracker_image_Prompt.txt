JavaScript:

const item = $input.item.json;

function findText(obj) {
  if (!obj || typeof obj !== 'object') return '';
  for (const key of Object.keys(obj)) {
    const val = obj[key];
    if (typeof val === 'string' && val.includes('"date"')) return val;
    if (typeof val === 'object') {
      const found = findText(val);
      if (found) return found;
    }
  }
  return '';
}

const raw = findText(item);
const cleaned = raw.replace(/```json|```/g, '').trim();

const match = cleaned.match(/\[[\s\S]*\]/);
if (!match) return [{ json: { error: 'No JSON found' } }];

let transactions;
try {
  transactions = JSON.parse(match[0]);
} catch(e) {
  return [{ json: { error: 'Parse failed' } }];
}

if (!Array.isArray(transactions)) {
  transactions = [transactions];
}

return transactions.map(data => ({
  json: {
    date: data.date || '',
    type: data.type ? data.type.toUpperCase() : 'EXPENSE',
    category: data.category || '',
    vendor: data.vendor || data.category || 'Not specified',
    platform: data.platform || 'Direct',
    product: data.product || 'Not specified',
    amount: Number(data.amount) || 0,
    currency: data.currency || 'INR',
    currency_symbol: data.currency_symbol || '₹'
  }
}));
